## 概述

`FulltextQueryer` 类负责构建高质量的全文检索查询，支持中英文混合查询、关键词权重计算、同义词扩展等功能。本文档详细解析全文查询的构建流程。

```python
class FulltextQueryer(QueryBase):
    """
    全文查询器

    提供全文检索查询功能，支持关键词提取、权重计算、
    同义词扩展等，用于构建高质量的检索查询。

    Attributes:
        tw: 词项权重计算器 (term_weight.Dealer)
        syn: 同义词查找器 (synonym.Dealer)
        query_fields: 查询字段列表，定义各字段的权重
    """

    def __init__(self):
        """初始化全文查询器"""
        self.tw = term_weight.Dealer()
        self.syn = synonym.Dealer()
        self.query_fields = [
            "title_tks^10",
            "title_sm_tks^5",
            "important_kwd^30",
            "important_tks^20",
            "question_tks^20",
            "content_ltks^2",
            "content_sm_ltks",
        ]
```

```python
def question(self, txt, tbl="qa", min_match: float = 0.6):
    """
    构建问题查询

    Args:
        txt: 用户问题文本
        tbl: 表名（默认 "qa"）
        min_match: 最小匹配阈值（默认 0.6）

    Returns:
        tuple: (MatchTextExpr, keywords)
            - MatchTextExpr: 匹配文本表达式
            - keywords: 提取的关键词列表
            - 如果无法处理则返回 (None, keywords)
    """
```

### 处理流程

```
question(txt, min_match=0.6)
    │
    ├─→ 【文本预处理】 :99-105
    │     ├─ add_space_between_eng_zh()  ← 中英文间加空格
    │     ├─ tradi2simp() / strQ2B()     ← 繁转简 / 全角转半角
    │     └─ rmWWW()                      ← 移除 URL
    │
    ├─→ 【非中文路径】 (英文等) :107-158
    │     ├─ rag_tokenizer.tokenize()    ← 分词
    │     ├─ tw.weights(tks)             ← 词项权重
    │     ├─ syn.lookup(tk)              ← 同义词扩展
    │     └─ 构建加权查询: "(word^0.5 syn1^0.1 syn2^0.1)"
    │
    └─→ 【中文路径】 :160-272
          ├─ tw.split(txt)               ← 中文分词
          ├─ tw.weights([tt])            ← 词项权重
          ├─ syn.lookup(tt)              ← 同义词扩展
          ├─ fine_grained_tokenize()     ← 细粒度分词
          └─ 构建查询: "(词^w OR 同义词^0.2 OR 细粒度~2^0.5)"
```

```python
if not self._is_chinese(txt):
    # 英文等语言处理
    tks = rag_tokenizer.tokenize(txt)      # 分词
    wts = self.tw.weights(tks)              # 计算词权重

    # 构建查询字符串
    qst = []
    for tk, wt in zip(tks, wts):
```

```python
# query.py:160-272
# 中文分词
tt = self.tw.split(txt)
tt_weights = self.tw.weights([tt])

# 构建查询
qst = []
for i, (t, w) in enumerate(zip(tt, tt_weights[0])):
    # 同义词扩展
    syns = self.syn.lookup(t)
    if syns:
        syn_str = " ".join([f"{s}^{w*0.2}" for s in syns])
        qst.append(f"({t}^{w} OR {syn_str})")
    else:
        qst.append(f"{t}^{w}")

    # 细粒度分词
    sm_tks = self.fine_grained_tokenize(t)
    if sm_tks:
        sm_str = " ".join([f"{s}^{w*0.5}" for s in sm_tks])
        qst.append(f'("{sm_str}"~2)^{w*0.5}')

query_string = " ".join(qst)
```

### 实现逻辑

```python
# query.py:276-309
def hybrid_similarity(self, avec, bvecs, atks, btkss, tkweight=0.3, vtweight=0.7):
    from sklearn.metrics.pairwise import cosine_similarity

    # 1. 向量相似度（余弦相似度）
    sims = cosine_similarity([avec], bvecs)[0]

    # 2. 词元相似度
    tksim = self.token_similarity(atks, btkss)

    # 3. 向量全零时降级为纯词元
    if np.sum(sims) == 0:
        return np.array(tksim), tksim, sims

    # 4. 加权混合
    hybrid_sim = vtweight * sims + tkweight * tksim

    return hybrid_sim, tksim, sims
```

### 混合策略

| 相似度类型 | 权重 | 计算方法 |
|-----------|------|---------|
| 向量相似度 | 0.7 | cosine_similarity |
| 词元相似度 | 0.3 | token_similarity (Jaccard + 位置权重) |

**说明**: 向量相似度占主导，词元相似度用于补充精确匹配。

## token_similarity() 方法详解

> 代码对齐说明：当前实现并未使用本文展示的 `_position_weight` 版本。  
> 实际代码是先将词元转换为加权字典（含相邻 bigram 权重），再调用 `similarity()` 计算分数。

**文件位置**: `rag/nlp/query.py:311-352`

### 方法签名

```python
def token_similarity(self, atks, btkss):
    """
    计算词元相似度

    Args:
        atks: 查询词元列表
        btkss: 文档词元列表列表

    Returns:
        list: 相似度分数列表
    """
```

### 实现逻辑

```python
# query.py:311-352
def token_similarity(self, atks, btkss):
    if not atks or not btkss:
        return [0.0] * len(btkss)

    atks_set = set(atks)
    scores = []

    for btk in btkss:
        if not btk:
            scores.append(0.0)
            continue

        btk_set = set(btk)

        # Jaccard 相似度
        intersection = atks_set & btk_set
        union = atks_set | btk_set
        jaccard = len(intersection) / len(union) if union else 0.0

        # 位置权重增强
        position_boost = self._position_weight(atks, btk)

        scores.append(jaccard * (1 + position_boost))

    return scores
```

### 位置权重

```python
# query.py:345-352
def _position_weight(self, atks, btk):
    """计算位置权重（词在前面权重更高）"""
    boost = 0.0
    for i, tk in enumerate(atks):
        if tk in btk:
            pos = btk.index(tk)
            # 位置越靠前，权重越高
            boost += (len(btk) - pos) / len(btk)

    return boost / len(atks) if atks else 0.0
```

## similarity() 方法详解

```python
def similarity(self, qtwt, dtwt):
    """
    Jaccard 相似度基础算法

    Args:
        qtwt: 查询词元权重字典 {token: weight}
        dtwt: 文档词元权重字典 {token: weight}

    Returns:
        float: 相似度分数
    """
```

### 实现逻辑

```python
# query.py:354-386
def similarity(self, qtwt, dtwt):
    if not qtwt or not dtwt:
        return 0.0

    # 计算交集
    intersection = set(qtwt.keys()) & set(dtwt.keys())

    if not intersection:
        return 0.0

    # 计算加权交集和
    numerator = sum(min(qtwt[tk], dtwt[tk]) for tk in intersection)

    # 计算加权并集和
    union = set(qtwt.keys()) | set(dtwt.keys())
    denominator = sum(max(qtwt.get(tk, 0), dtwt.get(tk, 0)) for tk in union)

    return numerator / denominator if denominator else 0.0
```

### 实现逻辑

```python
# query.py:388-421
def paragraph(self, content_tks, keywords):
    if not content_tks or not keywords:
        return None

    # 构建关键词查询
    keyword_queries = []
    for kw in keywords:
        if kw in content_tks:
            # 查找关键词位置
            positions = [i for i, tk in enumerate(content_tks) if tk == kw]

            # 提取上下文（前后各 5 个词）
            for pos in positions:
                start = max(0, pos - 5)
                end = min(len(content_tks), pos + 6)
                context = content_tks[start:end]

                # 构建短语查询
                phrase = " ".join(context)
                keyword_queries.append(f'"{phrase}"~10')

    if not keyword_queries:
        return None

    query_string = " OR ".join(keyword_queries)

    return MatchTextExpr(
        ["content_ltks"],
        query_string,
        100,
        {}
    )
```

```python
# query.py:423-430
def _is_chinese(self, txt):
    """判断文本是否主要为中文"""
    if not txt:
        return False
    chinese_chars = sum(1 for c in txt if '\u4e00' <= c <= '\u9fff')
    return chinese_chars / len(txt) > 0.3
```

```python
# query.py:432-445
def fine_grained_tokenize(self, token):
    """细粒度分词（将词进一步拆分）"""
    if not token:
        return []

    # 使用 rag_tokenizer 进行细粒度分词
    fine_tks = rag_tokenizer.tokenize(token, fine_grained=True)

    # 过滤停用词和单字
    fine_tks = [tk for tk in fine_tks if len(tk) > 1]

    return fine_tks
```
