## 概述

文档存储层是检索系统的底层抽象，屏蔽了不同存储引擎的差异。RAGFlow 支持多种存储引擎：

1. **Elasticsearch**: 分布式搜索和分析引擎
2. **Infinity**: 高性能向量数据库
3. **OceanBase**: 分布式关系数据库

本文档详细解析文档存储层的抽象接口和实现。

## DocStoreConnection 抽象接口

**文件位置**: `common/doc_store/doc_store_base.py:143-262`

### 接口定义

```python
class DocStoreConnection(ABC):
    """
    数据库操作抽象接口

    屏蔽不同存储引擎的差异，提供统一的检索接口。
    """

    @abstractmethod
    def db_type(self) -> str:
        """返回数据库类型"""

    @abstractmethod
    def search(
        self,
        index_names,      # 索引名列表
        highlight_fields, # 高亮字段列表
        filters,          # 过滤条件字典
        match_expressions,# 匹配表达式列表
        order_by_expr,    # 排序表达式
        req              # 原始请求
    ):
        """
        执行检索

        Returns:
            SearchResult: 检索结果对象
        """

    @abstractmethod
    def get(
        self,
        chunk_ids,       # chunk ID 列表
        fields          # 返回字段列表
    ):
        """
        获取指定 chunk 的内容

        Returns:
            list: chunk 字典列表
        """

    @abstractmethod
    def insert(
        self,
        chunk,           # chunk 字典
        chunk_id        # chunk ID
    ):
        """
        插入或更新 chunk
        """

    @abstractmethod
    def delete(
        self,
        chunk_ids       # chunk ID 列表
    ):
        """
        删除指定 chunk
        """
```

#### search() 实现

**文件位置**: `common/doc_store/es_conn_base.py`

```python
def search(
    self,
    index_names,
    highlight_fields,
    filters,
    match_expressions,
    order_by_expr,
    req
):
    """
    Elasticsearch 检索实现

    构建 ES DSL 查询并执行。
    """

    # 1. 构建 bool 查询
    bool_query = {
        "bool": {
            "must": [],
            "filter": []
        }
    }

    # 2. 添加过滤条件
    if filters.get("kb_id"):
        bool_query["bool"]["filter"].append({
            "terms": {"kb_id": filters["kb_id"]}
        })

    if filters.get("doc_id"):
        bool_query["bool"]["filter"].append({
            "terms": {"doc_id": filters["doc_id"]}
        })

    if filters.get("available_int"):
        bool_query["bool"]["filter"].append({
            "term": {"available_int": filters["available_int"]}
        })

    # 3. 添加匹配表达式
    for expr in match_expressions:
        if isinstance(expr, MatchTextExpr):
            # 全文匹配
            bool_query["bool"]["must"].append({
                "query_string": {
                    "fields": expr.fields,
                    "query": expr.matching_text,
                    "minimum_should_match": expr.extra_options.get("minimum_should_match", 0.6)
                }
            })

        elif isinstance(expr, MatchDenseExpr):
            # 向量匹配（需要 dense_vector 字段）
            bool_query["bool"]["must"].append({
                "knn": {
                    "field": expr.vector_column_name,
                    "query_vector": expr.embedding_data,
                    "k": expr.topn,
                    "num_candidates": expr.topn * 10
                }
            })

        elif isinstance(expr, FusionExpr):
            # 融合策略（在 ES 中通过多个查询组合实现）
            pass  # Fusion 在 ES 中通过查询组合实现

    # 4. 构建排序
    sort = []
    for field, order in order_by_expr.fields:
        sort.append({field: "desc" if order == 1 else "asc"})

    # 5. 构建高亮
    highlight = {}
    if highlight_fields:
        highlight["fields"] = {field: {} for field in highlight_fields}

    # 6. 执行查询
    response = self.client.search(
        index=index_names,
        body={
            "query": bool_query,
            "sort": sort,
            "highlight": highlight,
            "size": req.get("size", 10),
            "from": (req.get("page", 1) - 1) * req.get("size", 10)
        }
    )

    # 7. 解析结果
    return self._parse_search_result(response, req)
```

#### _parse_search_result() 实现

```python
def _parse_search_result(self, response, req):
    """解析 ES 检索结果"""

    hits = response["hits"]["hits"]

    # 提取结果
    ids = [hit["_id"] for hit in hits]
    total = response["hits"]["total"]["value"]

    # 提取字段
    field = {}
    for hit in hits:
        field[hit["_id"]] = hit["_source"]

    # 提取高亮
    highlight = {}
    if highlight_fields:
        for hit in hits:
            if "highlight" in hit:
                highlight[hit["_id"]] = hit["highlight"]

    # 提取聚合
    aggregation = self._parse_aggregation(response.get("aggregations", {}))

    return SearchResult(
        total=total,
        ids=ids,
        field=field,
        highlight=highlight,
        aggregation=aggregation
    )
```

#### search() 实现

**文件位置**: `common/doc_store/infinity_conn_base.py`

```python
def search(
    self,
    index_names,
    highlight_fields,
    filters,
    match_expressions,
    order_by_expr,
    req
):
    """
    Infinity 检索实现

    构建 Infinity 查询并执行。
    """

    # 1. 构建表名（Infinity 使用表而非索引）
    table_name = index_names[0]  # Infinity 单次查询只支持单表

    # 2. 构建匹配条件
    match_conditions = []

    for expr in match_expressions:
        if isinstance(expr, MatchTextExpr):
            # 全文匹配
            match_conditions.append({
                "type": "text",
                "fields": expr.fields,
                "matching_text": expr.matching_text,
                "topn": expr.topn
            })

        elif isinstance(expr, MatchDenseExpr):
            # 向量匹配
            match_conditions.append({
                "type": "dense",
                "vector_column": expr.vector_column_name,
                "embedding_data": expr.embedding_data,
                "embedding_type": expr.embedding_data_type,
                "distance_type": expr.distance_type,
                "topn": expr.topn
            })

        elif isinstance(expr, FusionExpr):
            # 融合策略
            match_conditions.append({
                "type": "fusion",
                "method": expr.method,
                "topn": expr.topn,
                "params": expr.fusion_params
            })

    # 3. 构建过滤条件
    filter_conditions = []
    if filters.get("kb_id"):
        filter_conditions.append({
            "field": "kb_id",
            "op": "in",
            "value": filters["kb_id"]
        })

    if filters.get("doc_id"):
        filter_conditions.append({
            "field": "doc_id",
            "op": "in",
            "value": filters["doc_id"]
        })

    # 4. 构建排序
    order_by = []
    for field, order in order_by_expr.fields:
        order_by.append({
            "field": field,
            "order": "desc" if order == 1 else "asc"
        })

    # 5. 执行查询
    response = self.client.search(
        table_name=table_name,
        match_conditions=match_conditions,
        filter_conditions=filter_conditions,
        order_by=order_by,
        limit=req.get("size", 10),
        offset=(req.get("page", 1) - 1) * req.get("size", 10)
    )

    # 6. 解析结果
    return self._parse_search_result(response, req)
```

#### Infinity 特性

| 特性 | 说明 |
|------|------|
| 内置归一化 | Infinity 返回的 _score 已归一化，无需额外处理 |

#### search() 实现

**文件位置**: `common/doc_store/ob_conn_base.py`

```python
def search(
    self,
    index_names,
    highlight_fields,
    filters,
    match_expressions,
    order_by_expr,
    req
):
    """
    OceanBase 检索实现

    构建 SQL 查询并执行。
    """

    # 1. 构建 SQL 查询
    sql = f"SELECT * FROM {index_names[0]}"

    # 2. 添加 WHERE 条件
    where_conditions = []

    # 过滤条件
    if filters.get("kb_id"):
        where_conditions.append(f"kb_id IN ({','.join(filters['kb_id'])})")

    if filters.get("doc_id"):
        where_conditions.append(f"doc_id IN ({','.join(filters['doc_id'])})")

    # 全文匹配（使用 FULLTEXT 索引）
    for expr in match_expressions:
        if isinstance(expr, MatchTextExpr):
            where_conditions.append(
                f"MATCH({expr.fields[0]}) AGAINST('{expr.matching_text}' IN BOOLEAN MODE)"
            )

    if where_conditions:
        sql += " WHERE " + " AND ".join(where_conditions)

    # 3. 添加排序
    order_by = []
    for field, order in order_by_expr.fields:
        order_by.append(f"{field} {'DESC' if order == 1 else 'ASC'}")

    if order_by:
        sql += " ORDER BY " + ", ".join(order_by)

    # 4. 添加分页
    sql += f" LIMIT {req.get('size', 10)} OFFSET {(req.get('page', 1) - 1) * req.get('size', 10)}"

    # 5. 执行查询
    cursor = self.conn.cursor(pymysql.cursors.DictCursor)
    cursor.execute(sql)
    results = cursor.fetchall()

    # 6. 解析结果
    return self._parse_search_result(results, req)
```

## 存储引擎对比

| 特性 | Elasticsearch | Infinity | OceanBase |
|------|---------------|----------|-----------|
| 全文检索 | ✅ 优秀 | ✅ 良好 | ⚠️ 基础 |
| 向量检索 | ✅ 良好 | ✅ 优秀 | ❌ 不支持 |
| 混合检索 | ✅ 支持 | ✅ 支持 | ❌ 不支持 |
| 聚合统计 | ✅ 优秀 | ✅ 良好 | ✅ 良好 |
| 分布式 | ✅ 支持 | ✅ 支持 | ✅ 支持 |
| 归一化分数 | ❌ 需要手动 | ✅ 内置 | - |
| 性能 | 中 | 高 | 高 |
| 适用场景 | 通用检索 | 向量检索 | 结构化数据 |

### ESConnectionPool

**文件位置**: `common/doc_store/es_conn_pool.py`

```python
class ESConnectionPool:
    """Elasticsearch 连接池"""

    _instance = None
    _lock = threading.Lock()

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance

    def get_connection(self):
        """获取 ES 连接"""
        if not hasattr(self, '_connection'):
            self._connection = ESConnection(
                host=settings.ES_HOST,
                port=settings.ES_PORT
            )
        return self._connection
```
