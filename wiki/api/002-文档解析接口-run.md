```python
async def run():
    req = await get_request_json()
    uid = current_user.id
    
    def _run_sync():
        # ========== 阶段1: 权限验证 ==========
        for doc_id in req["doc_ids"]:
            if not DocumentService.accessible(doc_id, uid):  # → 04️⃣
                return get_json_result(data=False, message="No authorization.")
        
        kb_table_num_map = {}
        
        # ========== 阶段2: 遍历处理每个文档 ==========
        for id in req["doc_ids"]:
            # 准备更新信息
            info = {"run": str(req["run"]), "progress": 0}
            if str(req["run"]) == TaskStatus.RUNNING.value and req.get("delete", False):
                info["progress_msg"] = ""
                info["chunk_num"] = 0
                info["token_num"] = 0
            
            # 获取租户和文档信息
            tenant_id = DocumentService.get_tenant_id(id)  # → 05️⃣
            if not tenant_id:
                return get_data_error_result(message="Tenant not found!")
            
            e, doc = DocumentService.get_by_id(id)  # → 06️⃣
            if not e:
                return get_data_error_result(message="Document not found!")
            
            # 处理取消操作
            if str(req["run"]) == TaskStatus.CANCEL.value:
                tasks = list(TaskService.query(doc_id=id))
                has_unfinished = any((t.progress or 0) < 1 for t in tasks)
                if str(doc.run) in [TaskStatus.RUNNING.value, TaskStatus.CANCEL.value] or has_unfinished:
                    cancel_all_task_of(id)  # → 07️⃣
                else:
                    return get_data_error_result(message="Cannot cancel a task not in RUNNING status")
            
            # 重新运行时清理计数
            if all([
                ("delete" not in req or req["delete"]),
                str(req["run"]) == TaskStatus.RUNNING.value,
                str(doc.run) == TaskStatus.DONE.value
            ]):
                DocumentService.clear_chunk_num_when_rerun(doc.id)  # → 08️⃣
            
            # 更新文档状态
            DocumentService.update_by_id(id, info)  # → 09️⃣
            
            # 删除已有数据（如果请求）
            if req.get("delete", False):
                TaskService.filter_delete([Task.doc_id == id])  # → 10️⃣
                if settings.docStoreConn.index_exist(search.index_name(tenant_id), doc.kb_id):
                    settings.docStoreConn.delete(
                        {"doc_id": id},
                        search.index_name(tenant_id),
                        doc.kb_id
                    )  # → 11️⃣
            
            # 启动新任务
            if str(req["run"]) == TaskStatus.RUNNING.value:
                # 应用知识库配置（如果请求）
                if req.get("apply_kb"):
                    e, kb = KnowledgebaseService.get_by_id(doc.kb_id)  # → 12️⃣
                    if not e:
                        raise LookupError("Can't find this dataset!")
                    doc.parser_config["llm_id"] = kb.parser_config.get("llm_id")
                    doc.parser_config["enable_metadata"] = kb.parser_config.get("enable_metadata", False)
                    doc.parser_config["metadata"] = kb.parser_config.get("metadata", {})
                    DocumentService.update_parser_config(doc.id, doc.parser_config)  # → 13️⃣
                
                doc_dict = doc.to_dict()
                DocumentService.run(tenant_id, doc_dict, kb_table_num_map)  # → 14️⃣
        
        return get_json_result(data=True)
    
    return await thread_pool_exec(_run_sync)  # 在线程池中执行
```

```python
@staticmethod
def accessible(doc_id, user_id):
    """检查用户是否有权限访问该文档"""
    e, doc = DocumentService.get_by_id(doc_id)
    if not e:
        return False
    
    # 检查用户是否是知识库所属租户的成员
    from api.db.services import UserTenantService
    tenants = UserTenantService.query(user_id=user_id)
    for tenant in tenants:
        if tenant.tenant_id == doc.tenant_id:
            return True
    return False
```

```python
    """通过文档 ID 查询所属租户 ID（通过 Knowledgebase 关联）"""
    docs = cls.model.select(Knowledgebase.tenant_id)/
        .join(Knowledgebase, on=(Knowledgebase.id == cls.model.kb_id))/
        .where(cls.model.id == doc_id, Knowledgebase.status == StatusEnum.VALID.value)/
        .dicts()
    
    if not docs:
        return None
    return docs[0]["tenant_id"]
```

```python
@classmethod
@DB.connection_context()
def clear_chunk_num_when_rerun(cls, doc_id):
    """重新运行文档解析时，将文档的 token/chunk 计数从知识库中扣减"""
    doc = cls.model.get_by_id(doc_id)
    assert doc, "Can't find document in database."
    
    num = (
        Knowledgebase.update(
            token_num=Knowledgebase.token_num - doc.token_num,
            chunk_num=Knowledgebase.chunk_num - doc.chunk_num,
        )
        .where(Knowledgebase.id == doc.kb_id)
        .execute()
    )
    return num
```

```python
@classmethod
def run(cls, tenant_id: str, doc: dict, kb_table_num_map: dict):
    """启动文档解析任务
    
    这是任务调度的核心方法，负责：
    1. 检查并清理 TABLE 类型解析器的 field_map
    2. 根据 pipeline_id 决定走 DataFlow 还是普通解析
    3. 创建任务记录并推入 Redis 队列
    """
    from api.db.services.task_service import queue_dataflow, queue_tasks
    from api.db.services.file2document_service import File2DocumentService
    
    doc["tenant_id"] = tenant_id
    doc_parser = doc.get("parser_id", ParserType.NAIVE)
    
    # 处理 TABLE 类型解析器的 field_map 清理
    if doc_parser == ParserType.TABLE:
        kb_id = doc.get("kb_id")
        if not kb_id:
            return
        if kb_id not in kb_table_num_map:
            count = DocumentService.count_by_kb_id(
                kb_id=kb_id,
                keywords="",
                run_status=[TaskStatus.DONE],
                types=[]
            )
            kb_table_num_map[kb_id] = count
            if kb_table_num_map[kb_id] <= 0:
                KnowledgebaseService.delete_field_map(kb_id)
    
    # 分支A: DataFlow 流程
    if doc.get("pipeline_id", ""):
        queue_dataflow(
            tenant_id,
            flow_id=doc["pipeline_id"],
            task_id=get_uuid(),
            doc_id=doc["id"]
        )  # → 15️⃣
    
    # 分支B: 普通解析任务
    else:
        bucket, name = File2DocumentService.get_storage_address(doc_id=doc["id"])
        queue_tasks(doc, bucket, name, 0)  # → 16️⃣
```

```python
def queue_dataflow(
    tenant_id: str,
    flow_id: str,
    task_id: str,
    doc_id: str = CANVAS_DEBUG_DOC_ID,
    file: dict = None,
    priority: int = 0,
    rerun: bool = False
) -> tuple[bool, str]:
    """创建 DataFlow 任务并推入 Redis 队列
    
    DataFlow 是 Agent Canvas 配置的工作流程
    """
    
    # 构建任务
    task = dict(
        id=task_id,
        doc_id=doc_id,
        from_page=0,
        to_page=100000000,
        task_type="dataflow" if not rerun else "dataflow_rerun",
        priority=priority,
        begin_at=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
    )
    
    # 删除旧任务（非调试/假文档）
    if doc_id not in [CANVAS_DEBUG_DOC_ID, GRAPH_RAPTOR_FAKE_DOC_ID]:
        TaskService.model.delete().where(TaskService.model.doc_id == doc_id).execute()
        DocumentService.begin2parse(doc_id)  # → 17️⃣
    
    # 插入任务记录
    bulk_insert_into_db(model=Task, data_source=[task], replace_on_conflict=True)
    
    # 补充任务信息
    task["kb_id"] = DocumentService.get_knowledgebase_id(doc_id)
    task["tenant_id"] = tenant_id
    task["dataflow_id"] = flow_id
    task["file"] = file
    
    # 推入 Redis 队列
    if not REDIS_CONN.queue_product(
        settings.get_svr_queue_name(priority),
        message=task
    ):
        return (False, "Can't access Redis. Please check the Redis' status.")
    
    return (True, task["id"])
```

```python
def queue_tasks(doc: dict, bucket: str, name: str, priority: int):
    """创建文档解析任务并推入 Redis 队列
    
    这是标准文档解析的任务创建方法，负责：
    1. 根据文档类型创建任务数组（PDF按页、Excel按行）
    2. 计算任务 digest 用于去重和优化
    3. 尝试复用之前任务的 chunks
    4. 将任务推入 Redis 队列
    """
    
    def new_task():
        return {
            "id": get_uuid(),
            "doc_id": doc["id"],
            "progress": 0.0,
            "from_page": 0,
            "to_page": 100000000,
            "begin_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        }
    
    parse_task_array = []
    
    # ========== 阶段1: 根据文档类型创建任务 ==========
    if doc["type"] == FileType.PDF.value:
        # PDF 按页范围分任务
        file_bin = settings.STORAGE_IMPL.get(bucket, name)
        do_layout = doc["parser_config"].get("layout_recognize", "DeepDOC")
        pages = PdfParser.total_page_number(doc["name"], file_bin)
        if pages is None:
            pages = 0
        
        page_size = doc["parser_config"].get("task_page_size") or 12
        if doc["parser_id"] == "paper":
            page_size = doc["parser_config"].get("task_page_size") or 22
        if doc["parser_id"] in ["one", "knowledge_graph"] /
            or do_layout != "DeepDOC" /
            or doc["parser_config"].get("toc_extraction", False):
            page_size = 10 ** 9
        
        page_ranges = doc["parser_config"].get("pages") or [(1, 10 ** 5)]
        for s, e in page_ranges:
            s -= 1
            s = max(0, s)
            e = min(e - 1, pages)
            for p in range(s, e, page_size):
                task = new_task()
                task["from_page"] = p
                task["to_page"] = min(p + page_size, e)
                parse_task_array.append(task)
    
    elif doc["parser_id"] == "table":
        # Excel 按行数分任务
        file_bin = settings.STORAGE_IMPL.get(bucket, name)
        rn = RAGFlowExcelParser.row_number(doc["name"], file_bin)
        for i in range(0, rn, 3000):
            task = new_task()
            task["from_page"] = i
            task["to_page"] = min(i + 3000, rn)
            parse_task_array.append(task)
    
    else:
        # 其他类型单个任务
        parse_task_array.append(new_task())
    
    # ========== 阶段2: 计算任务 digest ==========
    chunking_config = DocumentService.get_chunking_config(doc["id"])
    for task in parse_task_array:
        hasher = xxhash.xxh64()
        for field in sorted(chunking_config.keys()):
            if field == "parser_config":
                for k in ["raptor", "graphrag"]:
                    if k in chunking_config[field]:
                        del chunking_config[field][k]
            hasher.update(str(chunking_config[field]).encode("utf-8"))
        for field in ["doc_id", "from_page", "to_page"]:
            hasher.update(str(task.get(field, "")).encode("utf-8"))
        task["digest"] = hasher.hexdigest()
        task["progress"] = 0.0
        task["priority"] = priority
    
    # ========== 阶段3: 尝试复用之前任务的 chunks ==========
    prev_tasks = TaskService.get_tasks(doc["id"])
    ck_num = 0
    if prev_tasks:
        for task in parse_task_array:
            ck_num += reuse_prev_task_chunks(task, prev_tasks, chunking_config)
        TaskService.filter_delete([Task.doc_id == doc["id"]])
        
        # 删除旧的 chunks
        pre_chunk_ids = []
        for pre_task in prev_tasks:
            if pre_task["chunk_ids"]:
                pre_chunk_ids.extend(pre_task["chunk_ids"].split())
        if pre_chunk_ids:
            settings.docStoreConn.delete(
                {"id": pre_chunk_ids},
                search.index_name(chunking_config["tenant_id"]),
                chunking_config["kb_id"]
            )
    
    DocumentService.update_by_id(doc["id"], {"chunk_num": ck_num})
    
    # ========== 阶段4: 插入任务到数据库 ==========
    bulk_insert_into_db(Task, parse_task_array, True)
    DocumentService.begin2parse(doc["id"])  # → 17️⃣
    
    # ========== 阶段5: 将未完成的任务推入 Redis 队列 ==========
    unfinished_task_array = [task for task in parse_task_array if task["progress"] < 1.0]
    for unfinished_task in unfinished_task_array:
        assert REDIS_CONN.queue_product(
            settings.get_svr_queue_name(priority),
            message=unfinished_task
        ), "Can't access Redis. Please check the Redis' status."
```

```python
async def handle_task():
    """从队列获取并执行一个任务，处理完成或失败后确认消息"""
    global DONE_TASKS, FAILED_TASKS
    
    # 从 Redis 获取任务
    redis_msg, task = await collect()  # → 21️⃣
    if not task:
        await asyncio.sleep(5)
        return

    try:
        CURRENT_TASKS[task["id"]] = copy.deepcopy(task)
        await do_handle_task(task)  # → 22️⃣ 核心处理

        DONE_TASKS += 1
        CURRENT_TASKS.pop(task_id, None)
    except Exception as e:
        FAILED_TASKS += 1
        CURRENT_TASKS.pop(task_id, None)
        set_progress(task_id, prog=-1, msg=f"[Exception]: {str(e)}")

        pipeline_task_type = TASK_TYPE_TO_PIPELINE_TASK_TYPE.get(
            task_type, PipelineTaskType.PARSE
        ) or PipelineTaskType.PARSE
        if not task.get("dataflow_id", ""):
            PipelineOperationLogService.record_pipeline_operation(
                document_id=task["doc_id"],
                pipeline_id="",
                task_type=pipeline_task_type
            )
        
        # 确认消息（ACK）
        redis_msg.ack()
```

```python
async def collect():
    """从 Redis 队列获取任务
    
    Returns:
        元组 (redis_msg, task):
        - redis_msg: Redis 消息对象，用于后续 ACK 确认
        - task: 完整的任务字典
    
    若无可用任务或任务已取消，返回 (None, None)
    """
    global CONSUMER_NAME, UNACKED_ITERATOR
    
    svr_queue_names = settings.get_svr_queue_names()
    redis_msg = None
    
    try:
        # 尝试从未确认队列获取
        if not UNACKED_ITERATOR:
            UNACKED_ITERATOR = REDIS_CONN.get_unacked_iterator(
                svr_queue_names, SVR_CONSUMER_GROUP_NAME, CONSUMER_NAME
            )
        try:
            redis_msg = next(UNACKED_ITERATOR)
        except StopIteration:
            # 从队列获取新消息
            for svr_queue_name in svr_queue_names:
                redis_msg = REDIS_CONN.queue_consumer(
                    svr_queue_name, SVR_CONSUMER_GROUP_NAME, CONSUMER_NAME
                )
                if redis_msg:
                    break
    except Exception as e:
        logging.exception(f"collect got exception: {e}")
        return None, None
    
    if not redis_msg:
        return None, None
    
    msg = redis_msg.get_message()
    if not msg:
        logging.error(f"collect got empty message of {redis_msg.get_msg_id()}")
        redis_msg.ack()
        return None, None
    
    # 根据任务类型获取完整任务信息
    if msg.get("doc_id", "") in [GRAPH_RAPTOR_FAKE_DOC_ID, CANVAS_DEBUG_DOC_ID]:
        # GraphRAG 和 Canvas 调试任务
        task = msg
        if task["task_type"] in PIPELINE_SPECIAL_PROGRESS_FREEZE_TASK_TYPES:
            task = TaskService.get_task(msg["id"], msg["doc_ids"])
    elif msg.get("task_type") == PipelineTaskType.MEMORY.lower():
        # 记忆保存任务
        _, task_obj = TaskService.get_by_id(msg["id"])
        task = task_obj.to_dict()
    else:
        # 普通文档解析任务
        task = TaskService.get_task(msg["id"])
    
    # 检查任务是否已被取消
    if task:
        canceled = has_canceled(task["id"])
    if not task or canceled:
        state = "is unknown" if not task else "has been cancelled"
        FAILED_TASKS += 1
        logging.warning(f"collect task {msg['id']} {state}")
        redis_msg.ack()
        return None, None
    
    # 补充任务信息
    task_type = msg.get("task_type", "")
    task["task_type"] = task_type
    if task_type[:8] == "dataflow":
        task["tenant_id"] = msg["tenant_id"]
        task["dataflow_id"] = msg["dataflow_id"]
        task["kb_id"] = msg.get("kb_id", "")
    if task_type[:6] == "memory":
        task["memory_id"] = msg["memory_id"]
        task["source_id"] = msg["source_id"]
        task["message_dict"] = msg["message_dict"]
    
    return redis_msg, task
```

```python
async def build_chunks(task, progress_callback):
    """构建文档的 chunk（分块）数据，这是文档解析的核心流程
    
    完整处理流程：
    1. 检查文件大小是否超出限制
    2. 从 MinIO 获取文件二进制数据
    3. 调用对应格式的解析器（chunker）进行分块
    4. 将 chunk 中的图片上传到 MinIO，生成图片 ID
    5. 可选：LLM 自动生成关键词（auto_keywords）
    6. 可选：LLM 自动生成问题（auto_questions）
    7. 可选：LLM 自动生成元数据（enable_metadata）
    8. 可选：对 chunk 进行自动标签分类（tag_kb_ids）
    """

    bucket, name = File2DocumentService.get_storage_address(doc_id=task["doc_id"])
    file_bin = await get_storage_binary(bucket, name)  # → 24️⃣

        file_bin,
        parser_config=task["parser_config"],
        from_page=task["from_page"],
        to_page=task["to_page"],
        callback=progress_callback,
        tenant_id=task["tenant_id"],
        lang=task["language"]
    )

    for chunk in chunks:

    if parser_config.get("auto_keywords", 0):

            chunking_config, chunks, progress_callback
        )  # → 26️⃣

        chunks = await question_proposal(

            chunking_config, chunks, progress_callback
        )  # → 26️⃣
```

```python
async def embedding(chunks, embedding_model, parser_config, progress_callback):
    """对 chunks 进行向量嵌入
    
    批量处理 chunks，调用嵌入模型生成向量，并附加到 chunk
    """
    token_count = 0
    batch_size = 16
    
    # 分批处理
    for i in range(0, len(chunks), batch_size):
        batch = chunks[i:i+batch_size]
        
        # 提取文本
        texts = [chunk["content_with_weight"] for chunk in batch]
        token_count += sum(num_tokens_from_string(t) for t in texts)
        
        # 生成向量
        vectors, _ = await embedding_model.encode(texts)
        vector_size = len(vectors[0])
        
        # 将向量附加到 chunk
        for chunk, vector in zip(batch, vectors):
            chunk["q_%d_vec" % vector_size] = vector.tolist()
        
        # 更新进度
        progress_callback(
            msg="Embedding {}/{} chunks".format(i + len(batch), len(chunks))
        )
    
    return token_count, vector_size
```

## 📝 注意事项

### 1. 异步执行模型
- API 返回仅表示任务已创建，实际解析由后台 `task_executor` 进程执行
- 客户端需要轮询任务状态或通过 WebSocket 获取进度更新

### 2. Redis Stream 队列
- 任务通过 Redis Stream 进行异步通信
- 支持任务优先级（priority 参数）
- 支持任务去重（通过 digest）

### 3. 任务分片机制
- PDF 文件按页范围分任务（默认每 12 页一个任务）
- Excel 文件按行数分任务（每 3000 行一个任务）
- 其他文件类型单个任务

### 4. 任务复用优化
- 通过 digest 实现任务去重
- 相同配置的任务可以复用之前生成的 chunks
- 避免重复解析，提高效率

### 5. 并发控制
- `task_limiter` 控制最大并发任务数
- `kg_limiter` 控制 GraphRAG/RAPTOR 任务并发
- 避免资源耗尽

### 6. 超时机制
- 单个任务超时时间为 3 小时
- 超时后任务会被标记为失败

### 7. 取消机制
- 通过 Redis 设置取消标志（`{task_id}-cancel`）
- `task_executor` 定期检查取消标志
- 取消后自动清理已写入的数据

### 8. 错误处理
- 任务失败后记录错误信息到 `progress_msg`
- 失败任务不影响其他任务执行
- 支持重试机制
